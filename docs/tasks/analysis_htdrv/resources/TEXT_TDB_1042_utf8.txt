/*
TimeRec 구성(설치시작시간부터 초단위, 0.1초 정밀도)
①a,b: 차수 시작(a), 끝시간(b)
완착: 마지막 커넥터 장착됨
측정: 측정단계 진입함
탈거: 제품이 완전히 탈거됨
설치: 제품 설치가 처음 감지됨
바일치: 바코드 확인됨
바오류: 바코드 불일치
폐기: 폐기됨(완료 후에 일어남)
합격: 합격됨
바입력: 바코드가 읽힘
바출력: 바코드가 출력됨
바실패: 바코드 출력 실패
스탬프: 수탬프 출력
완료: 검사과정이 끝나고 결과를 기록함
긴합선: 합선한계시간 초과
에러기록: 검사 에러를 기록함

WHERE절은 기본 대소문자를 구분하지 않음
*/

CREATE TABLE Config (
	ID COUNTER NOT NULL,
	Contents LONGTEXT
);
CREATE UNIQUE INDEX PrimaryKey ON Config (ID ASC) WITH PRIMARY DISALLOW NULL;

CREATE TABLE Error (
	ID COUNTER NOT NULL,
	TestID INTEGER,
	Upload YESNO,
	Stage BYTE,
	SaveType BYTE,
	Type TEXT(1),
	Con SMALLINT,
	Pin SMALLINT,
	Io SMALLINT,
	Cir TEXT(20),
	Color TEXT(5),
	Width TEXT(5)
);
CREATE UNIQUE INDEX PrimaryKey ON Error (ID ASC) WITH PRIMARY DISALLOW NULL;
CREATE INDEX TestID ON Error (TestID ASC);

CREATE TABLE File (
	ID COUNTER NOT NULL,
	Path TEXT(255),
	FileName TEXT(255) NOT NULL,
	FileTime DATETIME NOT NULL,
	FileLength INTEGER NOT NULL,
	Exist YESNO,
	Read YESNO
);
CREATE UNIQUE INDEX PrimaryKey ON File (ID ASC) WITH PRIMARY DISALLOW NULL;

CREATE TABLE FileToPart (
	ID COUNTER NOT NULL,
	FileID INTEGER,
	PartID INTEGER
);
CREATE UNIQUE INDEX PrimaryKey ON FileToPart (ID ASC) WITH PRIMARY DISALLOW NULL;
CREATE INDEX FileToPartFileID ON FileToPart (FileID ASC);
CREATE INDEX PartID ON FileToPart (PartID ASC);

CREATE TABLE Part (ID COUNTER NOT NULL,
	PartName TEXT(20),
	PartNum TEXT(20),
	Eco TEXT(20),
	Nco TEXT(20),
	Alc TEXT(20),
	FirstUse DATETIME,
	LastUse DATETIME,
	TotalPass INTEGER,
	TotalFail INTEGER
);
CREATE UNIQUE INDEX PrimaryKey ON Part (ID ASC) WITH PRIMARY DISALLOW NULL;
CREATE INDEX PNum ON Part (PartNum ASC);

CREATE TABLE Prob (
	ID COUNTER NOT NULL,
	Criteria TEXT(255),
	Limit INTEGER,
	Count INTEGER
);
CREATE UNIQUE INDEX PrimaryKey ON Prob (ID ASC) WITH PRIMARY DISALLOW NULL;

CREATE TABLE Run (
	ID COUNTER NOT NULL,
	OnTime DATETIME,
	OffTime DATETIME,
	CNTVersion TEXT(4),
	HtdrvRelease TEXT(6),
	HtdrvName TEXT(30),
	WinVersion INTEGER,
	OptionHi INTEGER,
	OptionLo INTEGER,
	[Memo] LONGTEXT
);
CREATE UNIQUE INDEX PrimaryKey ON Run (ID ASC) WITH PRIMARY DISALLOW NULL;

CREATE TABLE Test (
	ID COUNTER NOT NULL,
	ServerID INTEGER,
	FileToPartID INTEGER,
	RunID INTEGER,
	ConfigID INTEGER,
	Upload YESNO,
	Lot TEXT(20),
	Pass INTEGER,
	Rework TEXT(5),
	Master TEXT(1),
	Result TEXT(1),
	StageProc TEXT(3),	
	Worker TEXT(20),
	TesterNum TEXT(20),
	TesterName TEXT(20),
	DataMatrix TEXT(30),
	LoadTime DATETIME,
	PassTime DATETIME,
	TimeRec LONGTEXT,
	BarRead LONGTEXT,
	BarWrite LONGTEXT,
	Res LONGTEXT,
	[Memo] LONGTEXT
);
CREATE UNIQUE INDEX PrimaryKey ON Test (ID ASC) WITH PRIMARY DISALLOW NULL;
CREATE INDEX ConfigID ON Test (ConfigID ASC);
CREATE INDEX RunID ON Test (RunID ASC);
CREATE INDEX ServerID ON Test (ServerID ASC);
CREATE INDEX StartTime ON Test (LoadTime ASC);

CREATE TABLE Worker (
	ID COUNTER NOT NULL,
	Code TEXT(50),
	Worker TEXT(50),
	Manager YESNO
);
CREATE UNIQUE INDEX PrimaryKey ON Worker (ID ASC) WITH PRIMARY DISALLOW NULL;
CREATE INDEX Code ON Worker (Code ASC);

ALTER TABLE Error ADD CONSTRAINT TestError FOREIGN KEY (TestID) REFERENCES Test (ID);
ALTER TABLE FileToPart ADD CONSTRAINT FileFileToPart FOREIGN KEY (FileID) REFERENCES File (ID);
ALTER TABLE FileToPart ADD CONSTRAINT PartFileToPart FOREIGN KEY (PartID) REFERENCES Part (ID);
ALTER TABLE Test ADD CONSTRAINT ConfigTest FOREIGN KEY (ConfigID) REFERENCES Config (ID);
ALTER TABLE Test ADD CONSTRAINT FileToPartTest FOREIGN KEY (FileToPartID) REFERENCES FileToPart (ID);
ALTER TABLE Test ADD CONSTRAINT RunTest FOREIGN KEY (RunID) REFERENCES Run (ID);
